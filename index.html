var FEUILLE_DONNEES = "Collecte"; var FEUILLE_POINTS = "Points"; function reponseJson(objet) { return ContentService.createTextOutput(JSON.stringify(objet)) .setMimeType(ContentService.MimeType.JSON); } // GET : ?action=points renvoie la liste des points function doGet(e) { var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : ""; if (action === "points") { return reponseJson({ ok: true, points: getPoints() }); } return reponseJson({ ok: true, message: "API de collecte opérationnelle" }); } // POST : enregistre une ligne dans la feuille "Collecte" function doPost(e) { try { var d = JSON.parse(e.postData.contents); var msg = enregistrerDonnees(d.badge, d.gps, d.bus, d.wilaya, d.daira, d.commune, d.nomPoint); return reponseJson({ ok: msg.indexOf("Succès") === 0, message: msg }); } catch (err) { return reponseJson({ ok: false, message: "Erreur : " + err }); } } function getPoints() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(FEUILLE_POINTS); if (!sheet || sheet.getLastRow() < 2) return []; var valeurs = sheet.getRange(2, 1, sheet.getLastRow() - 1, 5).getValues(); var points = []; valeurs.forEach(function(l) { var nom = String(l[3]).trim(); if (!nom) return; points.push({ wilaya: String(l[0]).trim(), daira: String(l[1]).trim(), commune: String(l[2]).trim(), nom: nom, gps: String(l[4]).trim() }); }); return points; } function enregistrerDonnees(badge, gpsPosition, bus, wilaya, daira, commune, nomPoint) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(FEUILLE_DONNEES); if (!sheet) sheet = ss.insertSheet(FEUILLE_DONNEES); if (sheet.getLastRow() === 0) { sheet.appendRow(["Horodatage", "Numéro de Badge", "Coordonnées GPS", "Numéro de Bus", "Wilaya", "Daira", "Commune", "Nom du Point"]); } sheet.appendRow([new Date(), badge, gpsPosition, bus || "", wilaya || "", daira || "", commune || "", nomPoint || ""]); return "Succès : Vos coordonnées de ramassage ont bien été enregistrées !"; } catch(e) { return "Erreur lors de l'écriture : " + e.toString(); } } // À exécuter UNE SEULE FOIS pour créer la feuille "Points" function creerFeuillePoints() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(FEUILLE_POINTS); if (!sheet) sheet = ss.insertSheet(FEUILLE_POINTS); if (sheet.getLastRow() === 0) { sheet.appendRow(["Wilaya", "Daira", "Commune", "Nom du Point", "Coordonnees_GPS"]); sheet.appendRow(["Oran", "Oran", "Oran", "Exemple - Place centrale", "35.6976, -0.6331"]); sheet.getRange("E:E").setNumberFormat("@"); sheet.getRange(1, 1, 1, 5).setFontWeight("bold"); sheet.setFrozenRows(1); } }
